set(ISF_APPLE_TEST_OUTPUT "${CMAKE_CURRENT_BINARY_DIR}")
set(ISF_APPLE_TEST_DATA "${CMAKE_SOURCE_DIR}/test_res" CACHE PATH "Apple integration-test fixtures")
set(ISF_APPLE_TEST_MODEL "${ISF_APPLE_TEST_DATA}/pack/Pikachu" CACHE FILEPATH "Apple integration-test model")
file(MAKE_DIRECTORY "${ISF_APPLE_TEST_OUTPUT}/output" "${ISF_APPLE_TEST_OUTPUT}/profiles/objc" "${ISF_APPLE_TEST_OUTPUT}/profiles/swift")
if(IOS)
    add_library(AppleAllocationProbe STATIC allocation_probe.c)
else()
    add_library(AppleAllocationProbe SHARED allocation_probe.c)
endif()
set_target_properties(AppleAllocationProbe PROPERTIES LIBRARY_OUTPUT_DIRECTORY "${ISF_APPLE_TEST_OUTPUT}" ARCHIVE_OUTPUT_DIRECTORY "${ISF_APPLE_TEST_OUTPUT}")
add_executable(AppleObjCContract objc_contract.m)
target_compile_options(AppleObjCContract PRIVATE -fobjc-arc)
if(ISF_SANITIZE_ADDRESS)
    target_compile_options(AppleObjCContract PRIVATE -fsanitize=address)
    target_link_options(AppleObjCContract PRIVATE -fsanitize=address)
endif()
set(ISF_APPLE_TEST_SWIFT_ARGS ${ISF_APPLE_SWIFT_SANITIZER_ARGS})
if(CMAKE_OSX_ARCHITECTURES)
    list(GET CMAKE_OSX_ARCHITECTURES 0 ISF_APPLE_TEST_ARCH)
    list(APPEND ISF_APPLE_TEST_SWIFT_ARGS --arch "${ISF_APPLE_TEST_ARCH}")
endif()
list(APPEND ISF_APPLE_TEST_SWIFT_ARGS --sdk "${ISF_APPLE_SDK}" "--minimum=${CMAKE_OSX_DEPLOYMENT_TARGET}")
if(ISF_ENABLE_APPLE_EXTENSION)
    target_compile_definitions(AppleObjCContract PRIVATE IF_TEST_COREML=1)
    list(APPEND ISF_APPLE_TEST_SWIFT_ARGS --coreml)
endif()
target_link_libraries(AppleObjCContract PRIVATE AppleAllocationProbe InspireFaceApple ${ISF_FOUNDATION} ${ISF_COREVIDEO})
set_target_properties(AppleObjCContract PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${ISF_APPLE_TEST_OUTPUT}")
add_custom_command(OUTPUT "${ISF_APPLE_TEST_OUTPUT}/AppleSwiftContract"
    COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/build_swift_test.py"
        --framework-dir "${ISF_APPLE_OUTPUT}" --source "${CMAKE_CURRENT_SOURCE_DIR}/swift_contract.swift"
        --output "${ISF_APPLE_TEST_OUTPUT}/AppleSwiftContract"
        ${ISF_APPLE_TEST_SWIFT_ARGS}
    DEPENDS InspireFaceSwift AppleAllocationProbe swift_contract.swift build_swift_test.py
    VERBATIM)
add_custom_target(AppleSwiftContractBuild ALL DEPENDS "${ISF_APPLE_TEST_OUTPUT}/AppleSwiftContract")
add_test(NAME Apple.API.Mapping COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/verify_api_coverage.py")
if(IOS)
    if(ISF_APPLE_SDK STREQUAL "iphoneos")
        # Device contracts are compile/link checks here; execution requires a signed host.
        return()
    endif()
    set(ISF_APPLE_TEST_LAUNCH ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/run_simulator.py")
endif()
add_test(NAME Apple.ObjC.Contract COMMAND ${ISF_APPLE_TEST_LAUNCH} $<TARGET_FILE:AppleObjCContract> "${ISF_APPLE_TEST_OUTPUT}/output")
add_test(NAME Apple.Swift.Contract COMMAND ${ISF_APPLE_TEST_LAUNCH} "${ISF_APPLE_TEST_OUTPUT}/AppleSwiftContract" "${ISF_APPLE_TEST_OUTPUT}/output")
if(EXISTS "${ISF_APPLE_TEST_MODEL}")
    add_test(NAME Apple.ObjC.Model COMMAND ${ISF_APPLE_TEST_LAUNCH} $<TARGET_FILE:AppleObjCContract> "${ISF_APPLE_TEST_OUTPUT}/output"
        "${ISF_APPLE_TEST_MODEL}" "${ISF_APPLE_TEST_DATA}/data/bulk/kun.jpg")
    add_test(NAME Apple.Swift.Model COMMAND ${ISF_APPLE_TEST_LAUNCH} "${ISF_APPLE_TEST_OUTPUT}/AppleSwiftContract" "${ISF_APPLE_TEST_OUTPUT}/output"
        "${ISF_APPLE_TEST_MODEL}" "${ISF_APPLE_TEST_DATA}/data/bulk/kun.jpg")
else()
    message(STATUS "Apple model tests not registered: ${ISF_APPLE_TEST_MODEL} is absent")
endif()
set_tests_properties(Apple.ObjC.Contract PROPERTIES ENVIRONMENT "LLVM_PROFILE_FILE=${ISF_APPLE_TEST_OUTPUT}/profiles/objc/%p-%m.profraw")
set_tests_properties(Apple.Swift.Contract PROPERTIES ENVIRONMENT "LLVM_PROFILE_FILE=${ISF_APPLE_TEST_OUTPUT}/profiles/swift/%p-%m.profraw")
if(EXISTS "${ISF_APPLE_TEST_MODEL}")
    set_tests_properties(Apple.ObjC.Model PROPERTIES ENVIRONMENT "LLVM_PROFILE_FILE=${ISF_APPLE_TEST_OUTPUT}/profiles/objc/%p-%m.profraw")
    set_tests_properties(Apple.Swift.Model PROPERTIES ENVIRONMENT "LLVM_PROFILE_FILE=${ISF_APPLE_TEST_OUTPUT}/profiles/swift/%p-%m.profraw")
endif()
